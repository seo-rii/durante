cmake_minimum_required(VERSION 3.16)
project(durante_guarded_sampler C)
find_package(DynamoRIO 11.3 EXACT REQUIRED)
set(previous "${CMAKE_CURRENT_SOURCE_DIR}/../../live_sampler_fast/probe")
function(require_sha path expected)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(SHA256 "${path}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Frozen sampler input changed: ${path}")
  endif()
endfunction()
set(base "${CMAKE_CURRENT_SOURCE_DIR}/../../access_sample/probe.c")
set(span_patch "${CMAKE_CURRENT_SOURCE_DIR}/../../cpp20_locality/probe/span.patch")
set(candidate_patch "${previous}/candidates.patch")
require_sha("${base}" "813817d50ac688895401cdda7f0d08a7a2f78cfca79e73fa61dbbd83ee79b503")
require_sha("${span_patch}" "e1c8114204bfd324d2a69b63baa51f74512c4a590d856113d431e88161f58fdf")
require_sha("${candidate_patch}" "0be08df6a4dfb291830812ae571f09e8a55032d16179f5398bce80399c2a713b")
require_sha("${previous}/profile.h" "20d81e09c4e235604d09baa5ba3aa336c3ac82007c9c1e37d9e1cdf13f39c52b")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The diagnostic guard requires Linux x86-64")
endif()
find_program(PATCH_TOOL patch)
if(NOT PATCH_TOOL)
  message(FATAL_ERROR "GNU patch is required")
endif()
set(span "${CMAKE_CURRENT_BINARY_DIR}/access_span.c")
set(candidate "${CMAKE_CURRENT_BINARY_DIR}/candidate.c")
set(source "${CMAKE_CURRENT_BINARY_DIR}/guarded.c")
add_custom_command(OUTPUT "${source}"
  COMMAND "${PATCH_TOOL}" --silent --batch --fuzz=0 --output "${span}" "${base}" "${span_patch}"
  COMMAND "${PATCH_TOOL}" --silent --batch --fuzz=0 --output "${candidate}" "${span}" "${candidate_patch}"
  COMMAND "${PATCH_TOOL}" --silent --batch --fuzz=0 --output "${source}" "${candidate}" "${CMAKE_CURRENT_SOURCE_DIR}/guarded.patch"
  DEPENDS "${base}" "${span_patch}" "${candidate_patch}" "${CMAKE_CURRENT_SOURCE_DIR}/guarded.patch"
  VERBATIM)
add_library(sampler_guarded SHARED "${source}")
configure_DynamoRIO_client(sampler_guarded)
use_DynamoRIO_extension(sampler_guarded drmgr)
target_compile_definitions(sampler_guarded PRIVATE SAMPLER_PROFILED)
target_include_directories(sampler_guarded PRIVATE "${previous}" "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(sampler_guarded PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
target_compile_options(sampler_guarded PRIVATE -Wall -Wextra -Werror)
