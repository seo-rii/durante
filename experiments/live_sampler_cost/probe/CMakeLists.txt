cmake_minimum_required(VERSION 3.16)
project(durante_sampler_cost C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The sampler cost experiment requires Linux x86-64")
endif()

find_package(DynamoRIO 11.3 EXACT REQUIRED)
find_program(PATCH_TOOL patch)
if(NOT PATCH_TOOL)
  message(FATAL_ERROR "GNU patch is required for the gate control")
endif()
set(V11_OBJECT "" CACHE FILEPATH "Unstripped durante.c.o from the frozen v11 release build")
set(SPAN_OBJECT "" CACHE FILEPATH "Unstripped access_span.c.o from the frozen full sampler build")

function(verify_hash path hash)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(SHA256 "${path}" actual)
  if(NOT actual STREQUAL hash)
    message(FATAL_ERROR "Wrong frozen input: ${path}")
  endif()
endfunction()

# Keep the exact original code generated by the two different compilers.
# Only the public initialization symbol needs a new name within this DSO.
function(rename_main input hash name)
  verify_hash("${input}" "${hash}")
  set(output "${CMAKE_CURRENT_BINARY_DIR}/${name}.o")
  add_custom_command(OUTPUT "${output}"
    COMMAND "${CMAKE_OBJCOPY}" --redefine-sym "dr_client_main=${name}" "${input}" "${output}"
    DEPENDS "${input}" VERBATIM)
  set_source_files_properties("${output}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
endfunction()

rename_main("${V11_OBJECT}" "897e8db7e89a22aa3129c08129426915777fa80589d8f46d8ac3dc1d701131fa" v11_main)
rename_main("${SPAN_OBJECT}" "295d76ce160bfe8b8a8b5bd947362dc726bf0860774caa0de56296e1f1885217" span_main)
add_library(sampler_fused SHARED fused.c "${CMAKE_CURRENT_BINARY_DIR}/v11_main.o" "${CMAKE_CURRENT_BINARY_DIR}/span_main.o")

set(previous "${CMAKE_CURRENT_SOURCE_DIR}/../../access_sample/probe.c")
set(span_patch "${CMAKE_CURRENT_SOURCE_DIR}/../../cpp20_locality/probe/span.patch")
set(gate_patch "${CMAKE_CURRENT_SOURCE_DIR}/gate.patch")
set(span_source "${CMAKE_CURRENT_BINARY_DIR}/access_span.c")
set(gate_source "${CMAKE_CURRENT_BINARY_DIR}/gate.c")
verify_hash("${previous}" "813817d50ac688895401cdda7f0d08a7a2f78cfca79e73fa61dbbd83ee79b503")
verify_hash("${span_patch}" "e1c8114204bfd324d2a69b63baa51f74512c4a590d856113d431e88161f58fdf")
add_custom_command(OUTPUT "${gate_source}"
  COMMAND "${PATCH_TOOL}" --silent --batch --fuzz=0 --output "${span_source}" "${previous}" "${span_patch}"
  COMMAND "${PATCH_TOOL}" --silent --batch --fuzz=0 --output "${gate_source}" "${span_source}" "${gate_patch}"
  DEPENDS "${previous}" "${span_patch}" "${gate_patch}" VERBATIM)
add_library(sampler_gate SHARED "${gate_source}")
add_library(sampler_empty SHARED empty.c)

foreach(client IN ITEMS sampler_fused sampler_gate sampler_empty)
  configure_DynamoRIO_client(${client})
  use_DynamoRIO_extension(${client} drmgr)
  set_target_properties(${client} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
  target_compile_options(${client} PRIVATE -Wall -Wextra -Werror)
endforeach()
