cmake_minimum_required(VERSION 3.16)
project(durante_access_sample C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The access sampler requires Linux x86-64")
endif()
find_package(DynamoRIO 11.3 EXACT REQUIRED)
add_library(access_sample SHARED probe.c)
configure_DynamoRIO_client(access_sample)
use_DynamoRIO_extension(access_sample drmgr)
set_target_properties(access_sample PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
target_compile_options(access_sample PRIVATE -Wall -Wextra -Werror)
