cmake_minimum_required(VERSION 3.16)
project(flat_cache_study C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The L1D diagnostic requires Linux x86-64")
endif()
find_package(DynamoRIO 11.3 EXACT REQUIRED)
add_library(durante_flat_cache SHARED ../../../client/durante.c)
configure_DynamoRIO_client(durante_flat_cache)
use_DynamoRIO_extension(durante_flat_cache drmgr)
set_target_properties(durante_flat_cache PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS YES)
target_compile_options(durante_flat_cache PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(durante_flat_cache PRIVATE DURANTE_TEST_FLAT_CACHE DURANTE_L1D_PROBE DURANTE_SDK_VERSION="11.3.0")
