cmake_minimum_required(VERSION 3.16)
project(graph_block_control C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The L1D diagnostic requires Linux x86-64")
endif()
find_package(DynamoRIO 11.3 EXACT REQUIRED)
add_library(durante_latency_callback SHARED ../../../client/durante.c)
configure_DynamoRIO_client(durante_latency_callback)
use_DynamoRIO_extension(durante_latency_callback drmgr)
set_target_properties(durante_latency_callback PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS YES)
target_compile_options(durante_latency_callback PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(durante_latency_callback PRIVATE DURANTE_TEST_LATENCY_CALLBACK DURANTE_L1D_PROBE DURANTE_SDK_VERSION="11.3.0")
