cmake_minimum_required(VERSION 3.16)
project(durante_temporal_sampler C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The temporal sampler requires Linux x86-64")
endif()
find_package(DynamoRIO 11.3 EXACT REQUIRED)
find_program(PATCH_TOOL patch REQUIRED)
set(base "${CMAKE_CURRENT_SOURCE_DIR}/../../access_sample/probe.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${base}")
file(SHA256 "${base}" digest)
if(NOT digest STREQUAL "813817d50ac688895401cdda7f0d08a7a2f78cfca79e73fa61dbbd83ee79b503")
  message(FATAL_ERROR "Unexpected registered base sampler")
endif()
set(source "${CMAKE_CURRENT_BINARY_DIR}/temporal.c")
add_custom_command(OUTPUT "${source}"
  COMMAND "${PATCH_TOOL}" --silent --batch --fuzz=0 --output "${source}" "${base}" "${CMAKE_CURRENT_SOURCE_DIR}/temporal.patch"
  DEPENDS "${base}" "${CMAKE_CURRENT_SOURCE_DIR}/temporal.patch" VERBATIM)
add_library(temporal SHARED "${source}")
configure_DynamoRIO_client(temporal)
use_DynamoRIO_extension(temporal drmgr)
set_target_properties(temporal PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
target_compile_options(temporal PRIVATE -Wall -Wextra -Werror -ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=.)
