cmake_minimum_required(VERSION 3.16)
project(durante_search_probe C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "search_probe requires Linux x86-64")
endif()
find_package(DynamoRIO 11.3 EXACT REQUIRED)
add_library(search_probe SHARED probe.c)
configure_DynamoRIO_client(search_probe)
use_DynamoRIO_extension(search_probe drmgr)
set_target_properties(search_probe PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
target_compile_options(search_probe PRIVATE -Wall -Wextra -Werror)
