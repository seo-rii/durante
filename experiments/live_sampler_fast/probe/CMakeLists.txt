cmake_minimum_required(VERSION 3.16)
project(durante_sampler_fast C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The live sampling candidates require Linux x86-64")
endif()
find_package(DynamoRIO 11.3 EXACT REQUIRED)
find_program(PATCH_TOOL patch)
if(NOT PATCH_TOOL)
  message(FATAL_ERROR "GNU patch is required to build the sampling candidates")
endif()

set(previous "${CMAKE_CURRENT_SOURCE_DIR}/../../access_sample/probe.c")
set(span_patch "${CMAKE_CURRENT_SOURCE_DIR}/../../cpp20_locality/probe/span.patch")
set(candidate_patch "${CMAKE_CURRENT_SOURCE_DIR}/candidates.patch")
set(span_source "${CMAKE_CURRENT_BINARY_DIR}/access_span.c")
set(source "${CMAKE_CURRENT_BINARY_DIR}/candidate.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${previous}" "${span_patch}")
file(SHA256 "${previous}" previous_hash)
file(SHA256 "${span_patch}" span_patch_hash)
if(NOT previous_hash STREQUAL "813817d50ac688895401cdda7f0d08a7a2f78cfca79e73fa61dbbd83ee79b503" OR
   NOT span_patch_hash STREQUAL "e1c8114204bfd324d2a69b63baa51f74512c4a590d856113d431e88161f58fdf")
  message(FATAL_ERROR "The registered full span sampler source has changed")
endif()
add_custom_command(OUTPUT "${source}"
  COMMAND "${PATCH_TOOL}" --silent --batch --fuzz=0 --output "${span_source}" "${previous}" "${span_patch}"
  COMMAND "${PATCH_TOOL}" --silent --batch --fuzz=0 --output "${source}" "${span_source}" "${candidate_patch}"
  DEPENDS "${previous}" "${span_patch}" "${candidate_patch}" VERBATIM)
add_custom_target(materialize_sampler DEPENDS "${source}")

foreach(mode IN ITEMS catalog packed profiled)
  set(client "sampler_${mode}")
  add_library(${client} SHARED "${source}")
  add_dependencies(${client} materialize_sampler)
  configure_DynamoRIO_client(${client})
  use_DynamoRIO_extension(${client} drmgr)
  string(TOUPPER "${mode}" upper)
  target_compile_definitions(${client} PRIVATE SAMPLER_${upper})
  target_include_directories(${client} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  set_target_properties(${client} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
  target_compile_options(${client} PRIVATE -Wall -Wextra -Werror)
endforeach()
