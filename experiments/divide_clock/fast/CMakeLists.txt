cmake_minimum_required(VERSION 3.16)
project(durante_divide_clock_fast C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The divide clock requires Linux x86-64")
endif()
find_package(DynamoRIO 11.3 EXACT REQUIRED)
add_library(divide_clock_fast SHARED fast.c)
configure_DynamoRIO_client(divide_clock_fast)
use_DynamoRIO_extension(divide_clock_fast drmgr)
target_include_directories(divide_clock_fast PRIVATE ../../../client)
set_target_properties(divide_clock_fast PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
target_compile_options(divide_clock_fast PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(divide_clock_fast PRIVATE DURANTE_SDK_VERSION="11.3.0")
