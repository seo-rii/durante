cmake_minimum_required(VERSION 3.16)
project(durante_divide_clock C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The divide-block study requires Linux x86-64")
endif()
find_package(DynamoRIO 11.3 EXACT REQUIRED)
add_library(divide_clock SHARED probe.c)
configure_DynamoRIO_client(divide_clock)
use_DynamoRIO_extension(divide_clock drmgr)
target_include_directories(divide_clock PRIVATE ../../client)
set_target_properties(divide_clock PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
target_compile_options(divide_clock PRIVATE -Wall -Wextra -Werror)
